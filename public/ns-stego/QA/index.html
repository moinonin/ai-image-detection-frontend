End-to-End QA Checklist¶

Run commands from repo root.

1) Start server

. .venv/bin/activate
PYTHONPATH=apps/api JWT_SECRET=dev-secret uvicorn server.app:app --port 8125

2) Generate test token

. .venv/bin/activate
python - <<'PY'
import jwt
print(jwt.encode({"sub":"tester","scopes":["provenance:embed","provenance:verify","crypto:encrypt","crypto:decrypt","keys:manage","keys:read","models:read","models:pull"]}, "dev-secret", algorithm="HS256"))
PY

3) Pull models

TOKEN=<paste>

curl -s -X POST http://127.0.0.1:8125/v1/models/pull \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"model_name":"meta-llama/Llama-3-8B"}'

curl -s -X POST http://127.0.0.1:8125/v1/models/pull \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"model_name":"sshleifer/tiny-gpt2"}'

4) Confirm health OK

curl -s http://127.0.0.1:8125/v1/health

5) LLM tiny model flow

TOKEN=<paste>

curl -s -X POST http://127.0.0.1:8125/v1/provenance/embed \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"secret":"Alice Johnson, 2026-02-12, CC-BY-NC","cover_prompt":"a short compliance memo","model_name":"sshleifer/tiny-gpt2","backend":"llm","bits_per_token":2,"max_new_tokens":400}'

6) Redundancy flow (MVP)

TOKEN=<paste>

curl -s -X POST http://127.0.0.1:8125/v1/provenance/embed \
  -H "Authorization: Bearer $TOKEN" \
  -H "Content-Type: application/json" \
  -d '{"secret":"Alice Johnson, 2026-02-12, CC-BY-NC","cover_prompt":"a short compliance memo","model_name":"meta-llama/Llama-3-8B","backend":"mvp","bits_per_token":2,"max_new_tokens":2000,"redundancy":3}'
